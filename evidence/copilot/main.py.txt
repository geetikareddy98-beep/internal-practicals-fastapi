from typing import List
from fastapi import FastAPI
from pydantic import BaseModel, Field, StrictStr


app = FastAPI(title="Teaching API", description="Small example for POST /items and GET /health")


class ItemCreate(BaseModel):
    """
    Request model for creating an item.


    - name: required string (StrictStr) with length between 1 and 100 characters.
      Whitespace is significant and preserved (the model does not strip or trim).
    """
    name: StrictStr = Field(
        ...,
        min_length=1,
        max_length=100,
        description="Item name (1-100 chars). Whitespace is preserved."
    )


class ItemResponse(BaseModel):
    """Response model for created item."""
    id: int
    name: str


@app.on_event("startup")
def _startup():
    """Initialize in-memory store on application startup."""
    app.state.items: List[dict] = []


@app.get("/health")
def health():
    """Health-check endpoint. Returns 200 and {"status": "ok"}."""
    return {"status": "ok"}


@app.post("/items", response_model=ItemResponse, status_code=201)
def create_item(item: ItemCreate):
    """
    Create an item and store it in memory.


    Behavior:
    - Validates `name` is a string (StrictStr) and has length 1..100.
    - Preserves leading/trailing/internal whitespace in `name`.
    - Stores items in app.state.items as dicts: {"id": int, "name": str}.
    - Returns 201 with the created item's id and name.
    - Invalid input (missing, too short/long, wrong type, or wrong body shape)
      results in a 422 validation response generated by FastAPI/Pydantic.
    """
    # Ensure store exists (defensive in case startup event didn't run)
    if not hasattr(app.state, "items"):
        app.state.items = []


    next_id = len(app.state.items) + 1
    stored = {"id": next_id, "name": item.name}
    app.state.items.append(stored)
    return stored
